# DSW France / TFR 2020 port
on_actions = {
	on_daily_FRA = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_country_flag = DSW_FRA_tfr_active
					}
				}
				if = {
					limit = {
						has_completed_focus = FRA_NEPAD_development
						NOT = {
							has_global_flag = AFRICA_nepad_focus
						}
						NOT = {
							has_country_flag = FRA_nepad_cleanup
						}
						has_dynamic_modifier = {
							modifier = FRA_idea_nepad_modifier
						}
					}
					set_country_flag = FRA_nepad_cleanup
					for_each_scope_loop = {
						array = global.is_african_nation
						if = {
							limit = {
								has_dynamic_modifier = {
									modifier = AFRICA_idea_nepad_modifier
								}
							}
							remove_dynamic_modifier = {
								modifier = AFRICA_idea_nepad_modifier
							}
						}
					}
					FRA = {
						remove_dynamic_modifier = {
							modifier = FRA_idea_nepad_modifier
						}
						clear_variable = FRA_nepad_sponsor_weekly_cost
					}
					clear_variable = global.africa_nepad_building_construction_modifier
					clear_variable = global.africa_nepad_education_modifier
					clear_variable = global.africa_nepad_social_care_modifier
					clear_variable = global.africa_nepad_health_cost_modifier
				}
			}
			if = {
				limit = {
					date > 2017.5.6
					date < 2020.1.1
					NOT = { has_country_flag = DSW_FRA_macron_2017_elected }
					NOT = { has_country_flag = DSW_FRA_tfr_active }
				}
				country_event = { id = dsw_fra.2017 }
			}
			if = {
				limit = {
					date > 2019.12.31
					NOT = { has_country_flag = DSW_FRA_tfr_active }
				}
				country_event = { id = dsw_fra.1 }
			}
			if = {
				limit = { has_country_flag = DSW_FRA_tfr_active }
				if = {
					limit = {
						date > 2020.1.15
						NOT = { has_country_flag = DSW_FRA_timeline_yellow_vests_pause }
					}
					set_country_flag = DSW_FRA_timeline_yellow_vests_pause
					country_event = { id = france.16 days = 1 }
				}
				if = {
					limit = {
						date > 2020.2.9
						NOT = { has_country_flag = DSW_FRA_timeline_covid }
					}
					set_country_flag = DSW_FRA_timeline_covid
					country_event = { id = france.15 days = 1 }
				}
				if = {
					limit = {
						date > 2020.3.22
						NOT = { has_country_flag = DSW_FRA_timeline_yellow_vests_return }
					}
					set_country_flag = DSW_FRA_timeline_yellow_vests_return
					country_event = { id = france.17 days = 1 }
				}
				if = {
					limit = {
						date > 2021.5.2
						NOT = { has_country_flag = DSW_FRA_timeline_military_demands }
					}
					set_country_flag = DSW_FRA_timeline_military_demands
					country_event = { id = france.5 days = 1 }
				}
				if = {
					limit = {
						date > 2022.3.1
						NOT = { has_country_flag = DSW_FRA_timeline_nfp_candidate }
					}
					set_country_flag = DSW_FRA_timeline_nfp_candidate
					country_event = { id = france.47 days = 1 }
				}
				if = {
					limit = {
						date > 2022.4.10
						NOT = { has_country_flag = DSW_FRA_timeline_election_2022 }
					}
					set_country_flag = DSW_FRA_timeline_election_2022
					country_event = { id = france.3 days = 1 }
				}
			}
		}
	}
	on_weekly_FRA = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_country_flag = DSW_FRA_tfr_active
					}
				}
				if = {
					limit = {
						has_completed_focus = FRA_the_space_program
					}
					add_to_variable = {
						FRA.FRA_peoples_support = -0.1
					}
					clamp_variable = {
						var = FRA.FRA_peoples_support
						min = 0
						max = 100
					}
				}
				clamp_variable = {
					var = FRA.FRA_peoples_support
					min = 0
					max = 100
				}
			}
		}
	}
}
