#       末尾给出 STUCK AT（当前卡在哪一步），由玩家自行决定是否继续。

#       （common/on_actions/00_on_actions.txt:107），无专属事件。

    # ------------------------------------------------------------------
    ZZZ_nc_ru_status = {
        icon = generic_political_actions
        cost = 0
        fire_only_once = no
        days_re_enable = 1

        allowed = {
            always = yes
        }

        visible = {
            is_ai = no
        }

        available = {
            is_ai = no
        }

        complete_effect = {
            custom_effect_tooltip = ZZZ_nc_ru_status_tt
            log = "[GetDateText]: ===== RU / NATO node check ====="

            # --- 条件 1：UKR 是否存在 ---
            if = {
                limit = {
                    country_exists = UKR
                }
                log = "[GetDateText]: CHECK country_exists UKR = YES"
            }
            else = {
                log = "[GetDateText]: CHECK country_exists UKR = NO"
            }

            # --- 条件 2：SOV 是否存在 ---
            if = {
                limit = {
                    country_exists = SOV
                }
                log = "[GetDateText]: CHECK country_exists SOV = YES"
            }
            else = {
                log = "[GetDateText]: CHECK country_exists SOV = NO"
            }

            # --- 条件 3：正式开战入口是否已执行（DSW_SOV_smo_started）---
            if = {
                limit = {
                    country_exists = SOV
                    SOV = { has_country_flag = DSW_SOV_smo_started }
                }
                log = "[GetDateText]: CHECK SOV flag DSW_SOV_smo_started = YES"
            }
            else = {
                log = "[GetDateText]: CHECK SOV flag DSW_SOV_smo_started = NO (DSW_SOV_declare_war_on_ukraine 未执行)"
            }

            # --- 条件 4：SOV 是否已与 UKR 交战 ---
            if = {
                limit = {
                    country_exists = SOV
                    country_exists = UKR
                    SOV = { has_war_with = UKR }
                }
                log = "[GetDateText]: CHECK SOV at war with UKR = YES"
            }
            else = {
                log = "[GetDateText]: CHECK SOV at war with UKR = NO"
            }

            # --- 条件 5：欧陆升级是否已开打 ---
            if = {
                limit = { has_global_flag = DSW_SOV_european_war_started }
                log = "[GetDateText]: CHECK DSW_SOV_european_war_started = YES"
            }
            else = {
                log = "[GetDateText]: CHECK DSW_SOV_european_war_started = NO"
            }

            if = {
                limit = { has_global_flag = ZZZ_eur_war_ongoing }
                log = "[GetDateText]: CHECK ZZZ_eur_war_ongoing = YES"
            }
            else = {
                log = "[GetDateText]: CHECK ZZZ_eur_war_ongoing = NO"
            }

            if = {
                limit = { has_global_flag = ZZZ_eur_war_resolved }
                log = "[GetDateText]: CHECK ZZZ_eur_war_resolved = YES"
            }
            else = {
                log = "[GetDateText]: CHECK ZZZ_eur_war_resolved = NO"
            }

            # --- 求值明细：卡在哪一步 ---
            if = {
                limit = {
                    NOT = { country_exists = UKR }
                }
                log = "[GetDateText]: STUCK AT: UKR does not exist"
            }
            else_if = {
                limit = {
                    NOT = { country_exists = SOV }
                }
                log = "[GetDateText]: STUCK AT: SOV does not exist"
            }
            else_if = {
                limit = {
                    NOT = { SOV = { has_country_flag = DSW_SOV_smo_started } }
                }
                log = "[GetDateText]: STUCK AT: official entry not executed (expect DSW_SOV_declare_war_on_ukraine)"
            }
            else_if = {
                limit = {
                    NOT = { SOV = { has_war_with = UKR } }
                }
                log = "[GetDateText]: STUCK AT: SOV not at war with UKR"
            }
            else_if = {
                limit = {
                    NOT = { has_global_flag = ZZZ_eur_war_ongoing }
                }
                log = "[GetDateText]: STUCK AT: europe war not escalated (expect ZZZ_eur_escalate)"
            }
            else_if = {
                limit = {
                    NOT = { has_global_flag = ZZZ_eur_war_resolved }
                }
                log = "[GetDateText]: STUCK AT: europe war ongoing, awaiting settlement"
            }
            else = {
                log = "[GetDateText]: STUCK AT: none, europe war resolved"
            }
        }

        ai_will_do = {
            base = 0
        }
    }
