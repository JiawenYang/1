	PRC_USA_RELATIONS_MAP_MODE = {
		bottom = {
			type = none
		}

			type = country
			color = {
				FROM = {
					if = {
						limit = { 
							OR = {
								original_tag = USA
								has_country_flag = PRC_USA_relations_stand_USA
						   	}
						}

						set_temp_variable = { green = 0 }
						set_temp_variable = { blue = 255 }
					}

						limit = { 
							OR = {
								original_tag = CHI

						   	}
						}

						set_temp_variable = { green = 189 }

		}

	}
