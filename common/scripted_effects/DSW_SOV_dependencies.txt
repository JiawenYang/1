DSW_SOV_dep_add_ruling_party_wing = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					political_power_ideology_var = ruling_party_wing_var_temp
				}
				is_in_array = {
					ruling_party_wings_array = ruling_party_wing_var_temp
				}
				AND = {
					has_government = var:ruling_party_wing_var_temp
					has_dynamic_modifier = {
						modifier = party_popularity_dynamic_modifier
					}
				}
			}
		}
		custom_effect_tooltip = add_ruling_party_wing
		hidden_effect = {
			set_temp_variable = {
				party_index = 20
			}
			set_temp_variable = {
				party_popularity_increase = 0
			}
			change_relative_party_popularity = yes
			recalculate_party = yes
		}
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					political_power_ideology_var = ruling_party_wing_var_temp
				}
				is_in_array = {
					coalition_partners_array = ruling_party_wing_var_temp
				}
				AND = {
					has_government = var:ruling_party_wing_var_temp
					has_dynamic_modifier = {
						modifier = party_popularity_dynamic_modifier
					}
				}
			}
		}
		set_variable = {
			is_in_coalition_with_@var:ruling_party_wing_var_temp = 1
		}
		add_to_array = {
			ruling_party_wings_array = ruling_party_wing_var_temp
		}
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = {
							modifier = party_popularity_dynamic_modifier_ruling_party_wings
						}
					}
				}
				add_dynamic_modifier = {
					modifier = party_popularity_dynamic_modifier_ruling_party_wings
				}
			}
		}
		add_to_variable = {
			ruling_party_wings_pp_gain = party_pop_array^ruling_party_wing_var_temp
		}
	}
	hidden_effect = {
		set_temp_variable = {
			var = ruling_popularity_var_temp
			value = 0.001
		}
		DSW_SOV_dep_add_ruling_party_popularity = yes
	}
	DSW_SOV_dep_update_party_popularity = yes
}

DSW_SOV_dep_get_current_government_type = {
	clear_variable = original_government_type
	set_variable = {
		original_government_type = current_party_ideology_group
	}
	if = {
		limit = {
			check_variable = {
				DSW_SOV_scripted_elections = 1
			}
		}
		set_country_flag = original_government_type_had_elections
	}
	DSW_SOV_dep_get_current_popularities = yes
	if = {
		limit = {
			has_coalition = yes
		}
		for_each_loop = {
			array = coalition_partners
			add_to_array = {
				original_coalitions = v
			}
		}
	}
	if = {
		limit = {
			has_autonomy_state = colonial_government
		}
		set_country_flag = colonial_government
	}
	else_if = {
		limit = {
			has_autonomy_state = integrated_puppet
		}
		set_country_flag = integrated_puppet
	}
	else_if = {
		limit = {
			has_autonomy_state = occupied_puppet
		}
		set_country_flag = occupied_puppet
	}
	else_if = {
		limit = {
			has_autonomy_state = autonomous_puppet
		}
		set_country_flag = autonomous_puppet
	}
	else_if = {
		limit = {
			has_autonomy_state = nominal_puppet
		}
		set_country_flag = nominal_puppet
	}
	else_if = {
		limit = {
			is_subject = yes
		}
		set_country_flag = default_puppet
	}
}
DSW_SOV_dep_remove_from_coalition = {
	if = {
		limit = {
			is_in_array = {
				coalition_partners_array = coalition_partner_var_temp
			}
		}
		custom_effect_tooltip = remove_from_coalition
		hidden_effect = {
			clear_variable = is_in_coalition_with_@var:coalition_partner_var_temp
			if = {
				limit = {
					has_dynamic_modifier = {
						modifier = party_popularity_dynamic_modifier_coalition
					}
					check_variable = {
						coalition_partners_array^num = 1
					}
				}
				remove_dynamic_modifier = {
					modifier = party_popularity_dynamic_modifier_coalition
				}
				clear_variable = coalition_pp_gain
				clear_array = coalition_partners_array
			}
			else = {
				remove_from_array = {
					coalition_partners_array = coalition_partner_var_temp
				}
				subtract_from_variable = {
					coalition_pp_gain = party_pop_array^coalition_partner_var_temp
				}
			}
		}
	}
	hidden_effect = {
		set_temp_variable = {
			var = ruling_popularity_var_temp
			value = -0.001
		}
		DSW_SOV_dep_add_ruling_party_popularity = yes
	}
	DSW_SOV_dep_update_party_popularity = yes
}

DSW_SOV_dep_add_to_coalition = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					political_power_ideology_var = coalition_partner_var_temp
				}
				is_in_array = {
					coalition_partners_array = coalition_partner_var_temp
				}
				AND = {
					has_government = var:coalition_partner_var_temp
					has_dynamic_modifier = {
						modifier = party_popularity_dynamic_modifier
					}
				}
			}
		}
		custom_effect_tooltip = add_to_coalition
		hidden_effect = {
			set_temp_variable = {
				party_index = 20
			}
			set_temp_variable = {
				party_popularity_increase = 0
			}
			change_relative_party_popularity = yes
			recalculate_party = yes
		}
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					political_power_ideology_var = coalition_partner_var_temp
				}
				is_in_array = {
					coalition_partners_array = coalition_partner_var_temp
				}
				AND = {
					has_government = var:coalition_partner_var_temp
					has_dynamic_modifier = {
						modifier = party_popularity_dynamic_modifier
					}
				}
			}
		}
		set_variable = {
			is_in_coalition_with_@var:coalition_partner_var_temp = 1
		}
		add_to_array = {
			coalition_partners_array = coalition_partner_var_temp
		}
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = {
							modifier = party_popularity_dynamic_modifier_coalition
						}
					}
				}
				add_dynamic_modifier = {
					modifier = party_popularity_dynamic_modifier_coalition
				}
			}
		}

			coalition_pp_gain = party_pop_array^coalition_partner_var_temp
		}

		set_temp_variable = {
			var = ruling_popularity_var_temp

				}

DSW_SOV_dep_update_party_popularity = {
	hidden_effect = {
		if = {
			limit = {
				has_variable = coalition_partners_array^num
			}
			set_variable = {
				coalition_pp_gain = 0
			}
			for_each_loop = {
				array = coalition_partners_array
				add_to_variable = {
					coalition_pp_gain = party_pop_array^v
				}
			}
			multiply_variable = {
				coalition_pp_gain = 0.5
			}
		}
		if = {
			limit = {
				has_variable = ruling_party_wings_array^num

			}

					ruling_party_wings_pp_gain = party_pop_array^v
				}
			}
			multiply_variable = {
				ruling_party_wings_pp_gain = 0.9
			}
		}
	}
