# ----------------------------------------------------------------------------
# ① 开战：ROOT = 发起方（SOV）
#   主对手按 美国 → 德国 → 法国 → 英国 的次序选定（首个存在且已入阵营者优先），
#   对该国 declare_war_on，其阵营成员由引擎 call-to-war 带入；
#   SOV 侧的阵营成员用 add_to_war 兜底拉入，与 DSW 的写法一致。
# ----------------------------------------------------------------------------
ZZZ_eur_begin_war = {
	if = {
		limit = {
			NOT = { has_global_flag = ZZZ_eur_war_ongoing }
			NOT = { has_global_flag = ZZZ_eur_war_resolved }
			NOT = { has_global_flag = DSW_SOV_european_war_started }
		}
		set_global_flag = ZZZ_eur_war_ongoing
		set_global_flag = ZZZ_eur_war_begins
		log = "[GetDateText]: ZZZ_eur_war_ongoing"

		# ---- 主对手：美国 ----
		if = {
			limit = {
				country_exists = USA
				USA = {
					is_in_faction = yes
					NOT = { is_in_faction_with = ROOT }
				}
			}
			USA = { set_country_flag = ZZZ_eur_nato_participant }
			USA = { set_country_flag = ZZZ_eur_nato_leader }
			declare_war_on = {
				target = USA
				type = annex_everything
			}
			every_country = {
				limit = {
					is_in_faction_with = USA
					NOT = { tag = ROOT }
				}
				set_country_flag = ZZZ_eur_nato_participant
			}
			every_country = {
				limit = {
					is_in_faction_with = ROOT
					NOT = { tag = ROOT }
					NOT = { has_war_with = USA }
				}
				add_to_war = {
					targeted_alliance = ROOT
					enemy = USA
					hostility_reason = asked_to_join
					single_target_only = no
				}
			}
		}
		# ---- 主对手：德国 ----
		else_if = {
			limit = {
				country_exists = GER
				GER = {
					is_in_faction = yes
					NOT = { is_in_faction_with = ROOT }
				}
			}
			GER = { set_country_flag = ZZZ_eur_nato_participant }
			GER = { set_country_flag = ZZZ_eur_nato_leader }
			declare_war_on = {
				target = GER
				type = annex_everything
			}
			every_country = {
				limit = {
					is_in_faction_with = GER
					NOT = { tag = ROOT }
				}
				set_country_flag = ZZZ_eur_nato_participant
			}
			every_country = {
				limit = {
					is_in_faction_with = ROOT
					NOT = { tag = ROOT }
					NOT = { has_war_with = GER }
				}
				add_to_war = {
					targeted_alliance = ROOT
					enemy = GER
					hostility_reason = asked_to_join
					single_target_only = no
				}
			}
		}
		# ---- 主对手：法国 ----
		else_if = {
			limit = {
				country_exists = FRA
				FRA = {
					is_in_faction = yes
					NOT = { is_in_faction_with = ROOT }
				}
			}
			FRA = { set_country_flag = ZZZ_eur_nato_participant }
			FRA = { set_country_flag = ZZZ_eur_nato_leader }
			declare_war_on = {
				target = FRA
				type = annex_everything
			}
			every_country = {
				limit = {
					is_in_faction_with = FRA
					NOT = { tag = ROOT }
				}
				set_country_flag = ZZZ_eur_nato_participant
			}
			every_country = {
				limit = {
					is_in_faction_with = ROOT
					NOT = { tag = ROOT }
					NOT = { has_war_with = FRA }
				}
				add_to_war = {
					targeted_alliance = ROOT
					enemy = FRA
					hostility_reason = asked_to_join
					single_target_only = no
				}
			}
		}
		# ---- 主对手：英国 ----
		else_if = {
			limit = {
				country_exists = ENG
				ENG = {
					is_in_faction = yes
					NOT = { is_in_faction_with = ROOT }
				}
			}
			ENG = { set_country_flag = ZZZ_eur_nato_participant }
			ENG = { set_country_flag = ZZZ_eur_nato_leader }
			declare_war_on = {
				target = ENG
				type = annex_everything
			}
			every_country = {
				limit = {
					is_in_faction_with = ENG
					NOT = { tag = ROOT }
				}
				set_country_flag = ZZZ_eur_nato_participant
			}
			every_country = {
				limit = {
					is_in_faction_with = ROOT
					NOT = { tag = ROOT }
					NOT = { has_war_with = ENG }
				}
				add_to_war = {
					targeted_alliance = ROOT
					enemy = ENG
					hostility_reason = asked_to_join
					single_target_only = no
				}
			}
		}
		# ---- 无任何西方目标：不开战，回滚旗标 ----
		else = {
			clr_global_flag = ZZZ_eur_war_ongoing
			clr_global_flag = ZZZ_eur_war_begins
			log = "[GetDateText]: ZZZ_eur_war aborted: no western target"
		}
	}
}

# ----------------------------------------------------------------------------
# ② 参战：标记双方参战方（在开战后调用，也可由 on_daily 补标）
#    ROOT = 发起方（SOV）
# ----------------------------------------------------------------------------
ZZZ_eur_mark_participants = {
	if = {
		limit = { has_global_flag = ZZZ_eur_war_ongoing }
		ROOT = { set_country_flag = ZZZ_eur_csto_participant }
		# 俄方：与 ROOT 同阵营者
		every_country = {
			limit = {
				NOT = { tag = ROOT }
				is_in_faction_with = ROOT
			}
			set_country_flag = ZZZ_eur_csto_participant
		}
		# 白俄罗斯：未入阵营则并入 ROOT 阵营，再标记
		if = {
			limit = {
				country_exists = BLR
				BLR = { NOT = { is_in_faction_with = ROOT } }
			}
			BLR = {
				if = {
					limit = { is_in_faction = yes }
					leave_faction = yes
				}
			}
			ROOT = { add_to_faction = BLR }
			BLR = { set_country_flag = ZZZ_eur_csto_participant }
		}
		# 西方：当前与 ROOT 交战的非俄方国家
		every_country = {
			limit = {
				NOT = { tag = ROOT }
				has_war_with = ROOT
				NOT = { has_country_flag = ZZZ_eur_csto_participant }
			}
			set_country_flag = ZZZ_eur_nato_participant
		}
	}
}

# ③ 结算前置：对全部参战方互相白和（只碰双方参战旗标持有者）
#    写法参照 DSW_SOV_conclude_european_war:792-824

ZZZ_eur_white_peace_all = {
	every_country = {
		limit = {
			OR = {
				has_country_flag = ZZZ_eur_csto_participant
				has_country_flag = ZZZ_eur_nato_participant
				# 上合合体后中国一侧（含东协成员与俄罗斯）
				has_country_flag = ZZZ_sco_merged_identity
			}
		}
		every_enemy_country = {
			limit = {
				OR = {
					has_country_flag = ZZZ_eur_csto_participant
					has_country_flag = ZZZ_eur_nato_participant
					# 上合合体后中国一侧（含东协成员与俄罗斯）
					has_country_flag = ZZZ_sco_merged_identity
				}
			}
			white_peace = PREV
		}
	}
}

ZZZ_eur_settle_nato_victory = {
	if = {
		limit = {
			has_global_flag = ZZZ_eur_war_ongoing
			NOT = { has_global_flag = ZZZ_eur_war_resolved }
		}
		# 已被吞并的东欧国家：先借 release_puppet 复国
		# 写法参照 DSW_SOV_campaign_effects.txt:1089-1102（is_core_of 判定 + release_puppet）
		if = {
			limit = {
				NOT = { country_exists = EST }
				any_owned_state = { is_core_of = EST }
			}
			release_puppet = EST
		}
		if = {
			limit = {
				NOT = { country_exists = LAT }
				any_owned_state = { is_core_of = LAT }
			}
			release_puppet = LAT
		}
		if = {
			limit = {
				NOT = { country_exists = LIT }
				any_owned_state = { is_core_of = LIT }
			}
			release_puppet = LIT
		}
		if = {
			limit = {
				NOT = { country_exists = POL }
				any_owned_state = { is_core_of = POL }
			}
			release_puppet = POL
		}
		if = {
			limit = {
				NOT = { country_exists = UKR }
				any_owned_state = { is_core_of = UKR }
			}
			release_puppet = UKR
		}
		# 波罗的海三国：先解除傀儡（release_puppet 由宗主国执行）
		if = {
			limit = { country_exists = EST EST = { is_puppet_of = SOV } }
			release_puppet = EST
		}
		if = {
			limit = { country_exists = LAT LAT = { is_puppet_of = SOV } }
			release_puppet = LAT
		}
		if = {
			limit = { country_exists = LIT LIT = { is_puppet_of = SOV } }
			release_puppet = LIT
		}
		# 波兰：脱离俄方，归还核心州
		if = {
			limit = { country_exists = POL POL = { is_puppet_of = SOV } }
			release_puppet = POL
		}
		if = {
			limit = { country_exists = POL }
			POL = { add_state_core = 1014 }
			POL = { add_state_core = 1098 }
			POL = { add_state_core = 111 }
			POL = { add_state_core = 112 }
			POL = { add_state_core = 113 }
			POL = { add_state_core = 114 }
			POL = { add_state_core = 115 }
			POL = { add_state_core = 116 }
			POL = { add_state_core = 433 }
			POL = { add_state_core = 474 }
			POL = { add_state_core = 625 }
			POL = { add_state_core = 626 }
			POL = { add_state_core = 628 }
			if = { limit = { 1014 = { NOT = { is_owned_by = POL } } } 1014 = { transfer_state_to = POL } }
			if = { limit = { 1098 = { NOT = { is_owned_by = POL } } } 1098 = { transfer_state_to = POL } }
			if = { limit = { 111 = { NOT = { is_owned_by = POL } } } 111 = { transfer_state_to = POL } }
			if = { limit = { 112 = { NOT = { is_owned_by = POL } } } 112 = { transfer_state_to = POL } }
			if = { limit = { 113 = { NOT = { is_owned_by = POL } } } 113 = { transfer_state_to = POL } }
			if = { limit = { 114 = { NOT = { is_owned_by = POL } } } 114 = { transfer_state_to = POL } }
			if = { limit = { 115 = { NOT = { is_owned_by = POL } } } 115 = { transfer_state_to = POL } }
			if = { limit = { 116 = { NOT = { is_owned_by = POL } } } 116 = { transfer_state_to = POL } }
			if = { limit = { 433 = { NOT = { is_owned_by = POL } } } 433 = { transfer_state_to = POL } }
			if = { limit = { 474 = { NOT = { is_owned_by = POL } } } 474 = { transfer_state_to = POL } }
			if = { limit = { 625 = { NOT = { is_owned_by = POL } } } 625 = { transfer_state_to = POL } }
			if = { limit = { 626 = { NOT = { is_owned_by = POL } } } 626 = { transfer_state_to = POL } }
			if = { limit = { 628 = { NOT = { is_owned_by = POL } } } 628 = { transfer_state_to = POL } }
		}
		# 乌克兰：脱离俄方控制
		if = {
			limit = { country_exists = UKR UKR = { is_puppet_of = SOV } }
			release_puppet = UKR
		}
		# 波罗的海核心州归还
		ZZZ_eur_restore_baltic_states = yes
		# 全员白和
		ZZZ_eur_white_peace_all = yes
		# 白和完成后再拆回合体（撤通行、清身份旗标）
		ZZZ_sco_split = yes

		clr_global_flag = ZZZ_eur_war_ongoing
		set_global_flag = ZZZ_eur_war_resolved
		log = "[GetDateText]: ZZZ_eur_nato_victory"
		every_country = {
			limit = { is_ai = no }
			news_event = { id = zzz_eur.2 days = 1 }
		}
	}
}

		# V4 党务重排：俄方胜利后重设东欧国家政权
		ZZZ_eur_settle_v4_governments = yes
		# 白和范围（精确到阵营名单，不用“所有与某国交战的国家”宽口径）
		if = {
			limit = {
				# 联动判定：东协（中国）与集安在同一场战争中，或两国已同阵营
				OR = {
					AND = {
						country_exists = USA
						SOV = { has_war_with = USA }
						CHI = { has_war_with = USA }
					}
					AND = {
						country_exists = GER
						SOV = { has_war_with = GER }
						CHI = { has_war_with = GER }
					}
					AND = {
						country_exists = FRA
						SOV = { has_war_with = FRA }
						CHI = { has_war_with = FRA }
					}
					AND = {
						country_exists = ENG
						SOV = { has_war_with = ENG }
						CHI = { has_war_with = ENG }
					}
					CHI = { is_in_faction_with = SOV }
				}
			}
			# 情形② 联动：除美国外的北约国家与集安白和，美国继续打
			every_country = {
				limit = {
					OR = {
						has_country_flag = ZZZ_eur_nato_participant
						has_idea = NATO_member
					}
					NOT = { original_tag = USA }
				}
				every_enemy_country = {
					limit = {
						OR = {
							original_tag = SOV
							has_country_flag = ZZZ_eur_csto_participant
							is_in_faction_with = SOV
							# 上合合体后中国一侧
							has_country_flag = ZZZ_sco_merged_identity
						}
					}
					white_peace = PREV
				}
			}
			# 清理参战标记：保留美国的北约参战旗标（美国继续与集安交战），其余清空
			every_country = {
				limit = { NOT = { original_tag = USA } }
				clr_country_flag = ZZZ_eur_nato_participant
				clr_country_flag = ZZZ_eur_csto_participant
				clr_country_flag = ZZZ_eur_nato_leader
			}
		}
		else = {
			# 情形① 未联动：北约全体（含美国）与集安白和
			every_country = {
				limit = {
					OR = {
						has_country_flag = ZZZ_eur_nato_participant
						has_idea = NATO_member
					}
				}
				every_enemy_country = {
					limit = {
						OR = {
							original_tag = SOV
							has_country_flag = ZZZ_eur_csto_participant
							is_in_faction_with = SOV
							# 上合合体后中国一侧
							has_country_flag = ZZZ_sco_merged_identity
						}
					}
					white_peace = PREV
				}
			}
			ZZZ_eur_clear_participants = yes
		}
		# 白和完成后再拆回合体（撤通行、清身份旗标）
		ZZZ_sco_split = yes
		# 白和已在上面按参战方逐对完成，不再置无效的 skip_default_capitulation
		set_global_flag = ZZZ_eur_csto_victory
		clr_global_flag = ZZZ_eur_war_ongoing
		set_global_flag = ZZZ_eur_war_resolved
		log = "[GetDateText]: ZZZ_eur_csto_victory"
		every_country = {
			limit = { is_ai = no }
			news_event = { id = zzz_eur.3 days = 1 }
		}
	}
}

# ③-e 双向联动（各自只跑一次）：
#    · 俄方先败、中国仍在战 → 蓝方欧洲（NATO 成员）加入对华战争
#    · 中国先败、俄方仍在战 → 蓝方亚洲（PDTO 成员）加入对俄战争
#    只用 add_to_war 建战争关系；不改阵营归属、不写 add_to_faction、不动 pdto_member；
#    已与该敌国交战者跳过；单点挂载在 on_daily。
# ----------------------------------------------------------------------------
ZZZ_eur_link_nato_vs_china = {
	if = {
		limit = {
			NOT = { has_global_flag = ZZZ_eur_link_nato_vs_china_done }
			OR = {
				has_global_flag = DSW_SOV_russian_defeat
				has_global_flag = ZZZ_eur_nato_victory
			}
			country_exists = CHI
			CHI = { has_war = yes }
			any_country = {
				OR = {
					has_idea = NATO_member
					has_country_flag = pdto_member
				}
				has_war_with = CHI
			}
		}
		set_global_flag = ZZZ_eur_link_nato_vs_china_done
		# 以“已在对华战争中的蓝方国家”为锚点，其余 NATO 成员用 add_to_war 加入同一场战争
		every_country = {
			limit = {
				OR = {
					has_idea = NATO_member
					has_country_flag = pdto_member
				}
				has_war_with = CHI
			}
			every_country = {
				limit = {
					has_idea = NATO_member
					NOT = { has_war_with = CHI }
				}
				add_to_war = {
					targeted_alliance = PREV
					enemy = CHI
					hostility_reason = asked_to_join
					single_target_only = no
				}
			}
		}
	}
}
