DSW_SOV_show_war_super_event = {
	if = {
		limit = {
			is_ai = no
			NOT = {
				has_global_flag = super_events_off_flag
			}
			NOT = {
				has_country_flag = DSW_SOV_war_super_seen
			}
		}
		set_country_flag = DSW_SOV_war_super_seen
		clr_country_flag = DSW_SOV_united_victory_super_event_flag
		set_country_flag = DSW_SOV_european_war_super_event_flag
		scoped_play_song = "DSW_SOV_european_war_score"
	}
}
