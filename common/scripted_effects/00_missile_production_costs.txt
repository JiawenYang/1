@missile_prod_math_max = 1000000
@missile_prod_weeks_max = 1000
@missile_prod_days_per_week = 7

space_auto_weekly_cost = {
	set_temp_variable = {
		temp_weekly = {
			value = temp_cost
			divide = {
				value = temp_days
				divide = @missile_prod_days_per_week
				clamp = { min = 1 max = @missile_prod_weeks_max }
			}
		}
	}
}

space_prod_weekly_expense = {

	space_prod_weekly_calc = yes
}
