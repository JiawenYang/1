# ----------------------------------------------------------------------------
# 韩国对北方开战（不可逆）
# 写法依据：events/05_china.txt:2321-2322（KOR 对 NKO 的先 create_wargoal 再 declare_war_on）
# 作用域：KOR
# ----------------------------------------------------------------------------
ZZZ_korea_declare_war_on_north = {
	if = {
		limit = {
			NKO = { exists = yes }
			NOT = { has_war_with = NKO }
		}
		create_wargoal = {
			type = annex_everything
			target = NKO
		}
		declare_war_on = {
			target = NKO
			type = annex_everything
		}
	}
}

# ----------------------------------------------------------------------------
# 由半岛侧引爆 GAW（大亚洲战争）
# 作用域：CHI（只能在中国作用域内调用）
#
# 调用顺序固定，禁止颠倒：
#   ① JAP 对 CHI 宣战（若已交战则跳过，避免重复宣战）
#   ② set_global_flag = ZZZ_gaw_started_by_korea
#   ③ PRC_gaw_start = yes
#   ④ 派发 prc.1008
#
# 为什么不能反：PRC_gaw_start 内部逐国调用 PRC_GAW_check_faction_members，
# 会按“当前阵营/交战状态”结算参战方与预备队 OOB。若先做初始化再宣战，
# 这一步会按空阵营（未交战）结算，导致日方参战方与 OOB 漏算、战争疲惫任务挂空。
# ----------------------------------------------------------------------------
ZZZ_start_gaw_from_korea = {
	if = {
		limit = {
			NOT = { has_global_flag = pan_asian_war_ongoing }
		}
		# ① 日本对华宣战（已交战则跳过）
		JAP = {
			if = {
				limit = {
					exists = yes
					NOT = { has_war_with = CHI }
				}
				declare_war_on = {
					target = CHI
					type = annex_everything
				}
			}
		}
		# ② 标记 GAW 由半岛侧引爆
		set_global_flag = ZZZ_gaw_started_by_korea
		# ③ GAW 全流程初始化（置 pan_asian_war_ongoing、加载 GAW 预备队 OOB、激活战争疲惫任务）
		PRC_gaw_start = yes
		# ④ 解锁中日双方 GAW 国策分支
		country_event = { id = prc.1008 hours = 6 }
	}
}
