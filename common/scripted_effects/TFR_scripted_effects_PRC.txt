#recruit character
PRC_recruit_character_effect = {
	if = {
		limit = {
			NOT = {
				THIS = { has_character_flag = is_dead }
				PLA = { has_character = PREV }
			}
		}
		if = {
			limit = { THIS = { is_army_leader = yes } }
			demote_to_general = yes
			remove_trait = { trait = recently_promoted }
		}
		set_nationality = CHI
	}
}
