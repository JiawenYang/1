### GAW（大亚洲战争）协作政府：快照 / 建立 / 恢复
### 千禧黎明 24 党派体系（party_pop_array^0..23），不照搬 TFR 的 10 意识形态键
### 调用约定：
###   ROOT = 目标国（被推掉的一方）
###   调用方先设置：set_variable = { var = gaw_collab_master value = <征服方 tag> }
###   可选旗标：gaw_collab_to_west（亲西方民主）/ gaw_collab_to_nationalist / gaw_collab_to_fascism（默认共产主义）
###   硬性例外：IND 与 SIA 按阵营分侧——日本侧（gaw_collab_to_west）走亲日民族主义 / 亲西方保守派，

###   通用旗标分支不得覆盖这两国的国别设定。
###   VIE 仅在 gaw_collab_to_west（日本侧）时变为南越伪政权；
###   中国侧 VIE 已从 on_capitulation 名单移除，完全不变
###   B 段要求：每国显式“换党 + 写支持率 + 安装具名领袖”，不只翻索引
###   越南共和国 cosmetic 复用 MD 既有 VIE_nationalist（旗为黄底三红条，已按 TGA 像素核验），
###   国号由本工程四语言本地化覆盖为“越南共和国”，不新建旗图文件。
### ============================================================

# 记录原执政党、选举状态、原附庸状态、24 党派支持率；只记录一次
ZZZ_gaw_snapshot_government = {
	if = {
		limit = { NOT = { has_country_flag = gaw_collaboration_snapshot_taken } }
		set_country_flag = gaw_collaboration_snapshot_taken
		# 原执政党（顶层意识形态，千禧可用的 5 键）
		if = { limit = { has_government = democratic } set_country_flag = gaw_orig_gov_democratic }
		else_if = { limit = { has_government = communism } set_country_flag = gaw_orig_gov_communism }
		else_if = { limit = { has_government = fascism } set_country_flag = gaw_orig_gov_fascism }
		else_if = { limit = { has_government = nationalist } set_country_flag = gaw_orig_gov_nationalist }
		else = { set_country_flag = gaw_orig_gov_neutrality }
		# 是否原本有选举
		if = { limit = { has_elections = yes } set_country_flag = gaw_orig_had_elections }
		# 是否原本就是附庸
		if = { limit = { is_subject = yes } set_country_flag = gaw_orig_was_subject }
		# 原执政党索引（0..23，用于精确恢复）
		set_variable = { gaw_orig_ruling_party_index = 0 }
		add_to_variable = { gaw_orig_ruling_party_index = var:ruling_party }
		# 原领袖快照：千禧领袖由 set_leader_<TAG> 按 ruling_party 确定性生成，
		# 故快照其所属党派索引即可在恢复时精确重建原领袖（不只回滚党派与支持率）
		set_variable = { gaw_orig_leader_party = 0 }
		add_to_variable = { gaw_orig_leader_party = var:ruling_party }
		set_country_flag = gaw_orig_leader_snapshot_taken
		# 24 党派支持率快照
		clear_array = gaw_orig_popularities
		add_to_array = { gaw_orig_popularities = party_pop_array^0 }
		add_to_array = { gaw_orig_popularities = party_pop_array^1 }

		add_to_array = { gaw_orig_popularities = party_pop_array^11 }
		add_to_array = { gaw_orig_popularities = party_pop_array^12 }
		add_to_array = { gaw_orig_popularities = party_pop_array^13 }
		add_to_array = { gaw_orig_popularities = party_pop_array^14 }
		add_to_array = { gaw_orig_popularities = party_pop_array^15 }
		add_to_array = { gaw_orig_popularities = party_pop_array^16 }

# 反推回来后恢复：解除傀儡、去掉 cosmetic、恢复原执政党与 24 党派支持率、清变量
ZZZ_gaw_restore_original_government = {
	if = {
		limit = { has_country_flag = gaw_collaboration_government }
		# 解除傀儡关系
		if = {
			limit = { has_variable = gaw_collab_master }
			if = {
				limit = { is_puppet_of = var:gaw_collab_master }
				var:gaw_collab_master = { end_puppet = ROOT }
			}
		}
		drop_cosmetic_tag = yes
		# 恢复原执政党
		if = {
			limit = { has_country_flag = gaw_orig_gov_democratic }
			if = {
				limit = { has_country_flag = gaw_orig_had_elections }
				set_politics = { ruling_party = democratic elections_allowed = yes }
			}
			else = { set_politics = { ruling_party = democratic elections_allowed = no } }
		}
		else_if = {
			limit = { has_country_flag = gaw_orig_gov_communism }
			if = {
				limit = { has_country_flag = gaw_orig_had_elections }
				set_politics = { ruling_party = communism elections_allowed = yes }
			}
			else = { set_politics = { ruling_party = communism elections_allowed = no } }
		}
		else_if = {
			limit = { has_country_flag = gaw_orig_gov_fascism }
			if = {
				limit = { has_country_flag = gaw_orig_had_elections }
				set_politics = { ruling_party = fascism elections_allowed = yes }
			}
			else = { set_politics = { ruling_party = fascism elections_allowed = no } }
		}
		else_if = {
			limit = { has_country_flag = gaw_orig_gov_nationalist }
			if = {
				limit = { has_country_flag = gaw_orig_had_elections }
				set_politics = { ruling_party = nationalist elections_allowed = yes }
			}
			else = { set_politics = { ruling_party = nationalist elections_allowed = no } }
		}
		else = {
			if = {
				limit = { has_country_flag = gaw_orig_had_elections }
				set_politics = { ruling_party = neutrality elections_allowed = yes }
			}
			else = { set_politics = { ruling_party = neutrality elections_allowed = no } }
		}
		# 恢复 24 党派支持率
		if = {
			limit = { has_variable = gaw_orig_popularities }
			set_variable = { party_pop_array^0 = gaw_orig_popularities^0 }
			set_variable = { party_pop_array^1 = gaw_orig_popularities^1 }
			set_variable = { party_pop_array^2 = gaw_orig_popularities^2 }
			set_variable = { party_pop_array^3 = gaw_orig_popularities^3 }
			set_variable = { party_pop_array^4 = gaw_orig_popularities^4 }
			set_variable = { party_pop_array^5 = gaw_orig_popularities^5 }
			set_variable = { party_pop_array^6 = gaw_orig_popularities^6 }
			set_variable = { party_pop_array^7 = gaw_orig_popularities^7 }
			set_variable = { party_pop_array^8 = gaw_orig_popularities^8 }
			set_variable = { party_pop_array^9 = gaw_orig_popularities^9 }
			set_variable = { party_pop_array^10 = gaw_orig_popularities^10 }
			set_variable = { party_pop_array^11 = gaw_orig_popularities^11 }
			set_variable = { party_pop_array^12 = gaw_orig_popularities^12 }
			set_variable = { party_pop_array^13 = gaw_orig_popularities^13 }
			set_variable = { party_pop_array^14 = gaw_orig_popularities^14 }
			set_variable = { party_pop_array^15 = gaw_orig_popularities^15 }
			set_variable = { party_pop_array^16 = gaw_orig_popularities^16 }
			set_variable = { party_pop_array^17 = gaw_orig_popularities^17 }
			set_variable = { party_pop_array^18 = gaw_orig_popularities^18 }
			set_variable = { party_pop_array^19 = gaw_orig_popularities^19 }
			set_variable = { party_pop_array^20 = gaw_orig_popularities^20 }
			set_variable = { party_pop_array^21 = gaw_orig_popularities^21 }
			set_variable = { party_pop_array^22 = gaw_orig_popularities^22 }
			set_variable = { party_pop_array^23 = gaw_orig_popularities^23 }
			clear_array = gaw_orig_popularities
		}
		# 精确恢复原执政党索引
		if = {
			limit = { has_variable = gaw_orig_ruling_party_index }
			set_variable = { ruling_party = var:gaw_orig_ruling_party_index }
			clear_variable = gaw_orig_ruling_party_index
		}
		# 精确恢复原领袖所属党派（优先用领袖快照字段，回落原执政党索引）
		if = {
			limit = { has_variable = gaw_orig_leader_party }
			set_variable = { ruling_party = var:gaw_orig_leader_party }
			clear_variable = gaw_orig_leader_party
		}
		# 恢复后只刷新联盟强度：不再调用 recalculate_party，
		# 因其内部 add_up_party_totals / set_party_positions 会对刚还原的 24 党值二次归一化并重建位置。

		clr_country_flag = gaw_orig_was_subject
		clr_country_flag = gaw_orig_leader_snapshot_taken
		clr_country_flag = gaw_collab_to_west
		clr_country_flag = gaw_collab_to_nationalist
		clr_country_flag = gaw_collab_to_fascism
		clr_country_flag = zzz_gaw_south_vietnam
		clear_variable = gaw_collab_master
		clear_variable = gaw_collab_old_master
	}
}
