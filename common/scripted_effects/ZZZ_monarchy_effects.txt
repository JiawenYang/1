# 配套文件：
#   common/characters/ZZZ_monarchs_europe.txt            （欧洲 8 位君主，5 大组角色全覆盖）
#   common/characters/ZZZ_monarchs_asia_mea.txt          （亚洲/中东/非洲/大洋洲 16 位君主）
#   common/scripted_effects/ZZZ_monarchy_crown_effects.txt（逐国 ZZZ_crown_<TAG> 加冕）

# 工作方式（与 TFR 同构，TFR 见 3350890356/common/characters/TFR_characters_ENG.txt:8-96）：
#   1) 开局落地：39 国的 2017 history 块（history/countries/ZZ9 - 2017.txt）直接
#      set_country_flag = ZZZ_monarchy_active + recruit_character + promote_character
#      （5 个意识形态大组各一次），不经过任何 on_action / scripted_effect；
#      history 只能调用内建效果，故此处只保留运行期使用的守护与废除效果；
#   2) set_leader 总闸门（common/scripted_effects/00_leader_setup_scripted_effects.txt:9）
#      只跳过 kill_country_leader / set_leader_[TAG] / set_correct_ideology_trait，
#      update_set_politics 照常执行，选举结果仍会写入执政大组；
#   3) on_government_change（common/on_actions/00_on_actions.txt:1369）调 ZZZ_monarch_guard，
#      君主制仍在时重新加冕；一旦 ZZZ_monarchy_abolished 生效就停手。
#
# 实证：MD 自己的注释已确认「子意识形态互换不改变大组，on_government_change 不会触发」
#   -> 394360/2777392649/common/scripted_effects/00_MD_politicsview_scripted_effects.txt:2045
#   引擎的党首是「按大组」存储的（TFR 的伊丽莎白在每个大组只挂 1 块也能跨子意识
#   形态一直当元首，见 3350890356/common/characters/TFR_characters_ENG.txt:8-96），
#   因此子意识形态层级的换党由"每个大组 1 角色 + promote 全 5 组"兜住，不依赖钩子。
# ============================================================================


# ----------------------------------------------------------------------------
# 换政府守护：君主制仍在时，若君主角色被移除（kill_country_leader / retire 等），
# 重新 recruit 并把 5 个党的党首身份重新扶正；一旦君主制废除旗标生效则完全不动作。

# ----------------------------------------------------------------------------
# 废除君主制：任何剧情/决议改共和时调用即可。
# 调用后 ZZZ_active_monarchy 失效，set_leader 恢复正常，元首交回政党指派。
# ----------------------------------------------------------------------------
ZZZ_abolish_monarchy = {
	set_country_flag = ZZZ_monarchy_abolished
	clr_country_flag = ZZZ_monarchy_active
}

# ----------------------------------------------------------------------------
# 复辟君主制：清掉废除旗标、恢复 active，并立即按当前旗标/日期选出现实在位
# 君主。ZZZ_crown_<TAG> 的日期兜底保证复辟国家拿到的是「当时在位」的那位，
# 而不是 2017 开局那位（例：2023 年复辟的加拿大拿到查尔斯三世而非伊丽莎白二世）。
# ----------------------------------------------------------------------------
ZZZ_restore_monarchy = {
	clr_country_flag = ZZZ_monarchy_abolished
	set_country_flag = ZZZ_monarchy_active
	ZZZ_monarch_guard = yes
}
