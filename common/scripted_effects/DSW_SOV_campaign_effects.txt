DSW_SOV_begin_european_war = {
	if = {
		limit = {
			NOT = {
				has_global_flag = DSW_SOV_european_war_started
			}
			NOT = {
				has_global_flag = DSW_SOV_european_war_resolved
			}
			NOT = {
				has_global_flag = ZZZ_eur_war_ongoing
			}
		}
		set_global_flag = DSW_SOV_european_war_started
		log = "[GetDateText]: DSW_SOV_european_war_started"
		set_global_flag = SOV_nato_war_begins_global
		if = {
			limit = {
				USA = {
					is_in_faction = yes
				}
				GER = {
					is_in_faction_with = USA
				}
			}
			USA = {
				every_country = {
					limit = {
						is_faction_leader = yes
						OR = {
							tag = USA
							is_in_faction_with = USA
						}
					}
					add_to_faction = UKR
				}
			}
		}
		else_if = {
			limit = {
				GER = {
					is_in_faction = yes
				}
			}
			GER = {
				every_country = {
					limit = {
						is_faction_leader = yes
						OR = {
							tag = GER
							is_in_faction_with = GER
						}
					}
					add_to_faction = UKR
				}
			}
		}
		else_if = {
			limit = {
				FRA = {
					is_in_faction = yes
				}
			}
			FRA = {
				every_country = {
					limit = {
						is_faction_leader = yes
						OR = {
							tag = FRA
							is_in_faction_with = FRA
						}
					}
					add_to_faction = UKR
				}
			}
		}
		else = {
			GER = {
				create_faction = NATO
				add_to_faction = UKR
			}
		}
		# 美国入欧战正式入口：不依赖内战，仅看 USA 存在、未处于战争、且尚未参战
		if = {
			limit = {
				country_exists = USA
				USA = {
					has_war = no
					NOT = { is_in_faction_with = UKR }
					NOT = { has_country_flag = DSW_SOV_nato_war_participant }
				}
			}
			every_country = {
				limit = {
					is_faction_leader = yes
					is_in_faction_with = UKR
				}
				add_to_faction = USA
			}
		}
		UKR = {
			set_country_flag = NATO_member
			set_country_flag = DSW_SOV_nato_war_participant
			remove_ideas = CSTO_member
			remove_from_tech_sharing_group = CSTO_Tech_Share
			add_ideas = NATO_member
			add_to_tech_sharing_group = NATO_Tech_Share
		}
		DSW_SOV_join_belarus_after_nato = yes
		DSW_SOV_open_regional_fronts = yes
		every_country = {
			limit = {
				is_in_faction_with = UKR
				NOT = {
					tag = UKR
				}
			}
			set_country_flag = DSW_SOV_nato_war_participant
			if = {
				limit = {
					NOT = {
						has_war_with = SOV
					}
				}
				add_to_war = {
					targeted_alliance = UKR
					enemy = SOV
					hostility_reason = asked_to_join
					single_target_only = no
				}
			}
		}
		every_country = {
			limit = {
				is_in_faction_with = SOV
				NOT = {
					tag = SOV
				}
			}
			set_country_flag = DSW_SOV_csto_war_participant
			if = {
				limit = {
					NOT = {
						has_war_with = UKR
					}
				}
				add_to_war = {
					targeted_alliance = SOV
					enemy = UKR
					hostility_reason = asked_to_join
					single_target_only = no
				}
			}
		}
		every_country = {
			limit = {
				is_ai = no
			}
			log = "DSW: first European war player notification"
			DSW_SOV_show_war_super_event = yes
			country_event = {
				id = dsw_sov.40
				days = 1
			}
		}
		SOV = {
			activate_mission = DSW_SOV_european_war_watch
		}
	}
}
