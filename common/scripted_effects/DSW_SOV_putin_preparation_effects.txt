DSW_SOV_integrate_novorossiya_effect = {
	every_state = {
		limit = {
			OR = {
				is_core_of = NOV
				is_core_of = DPR
				is_core_of = LPR
			}
		}
		add_core_of = SOV
	}
	if = {
		limit = {
			country_exists = NOV
			NOT = {
				has_war_with = NOV
			}
		}
		annex_country = {
			target = NOV
			transfer_troops = yes
		}
	}
	if = {
		limit = {
			country_exists = DPR
			NOT = {
				has_war_with = DPR
			}
		}
		annex_country = {
			target = DPR
			transfer_troops = yes
		}
	}
	if = {
		limit = {
			country_exists = LPR
			NOT = {
				has_war_with = LPR
			}
		}
		annex_country = {
			target = LPR
			transfer_troops = yes
		}
	}
	set_country_flag = DSW_SOV_novorossiya_integrated
}
