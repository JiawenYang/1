DSW_FRA_dep_remove_from_coalition = {
	if = {
		limit = {
			is_in_array = {
				coalition_partners_array = coalition_partner_var_temp
			}
		}
		custom_effect_tooltip = remove_from_coalition
		hidden_effect = {
			clear_variable = is_in_coalition_with_@var:coalition_partner_var_temp
			if = {
				limit = {
					has_dynamic_modifier = {
						modifier = party_popularity_dynamic_modifier_coalition
					}
					check_variable = {
						coalition_partners_array^num = 1
					}
				}
				remove_dynamic_modifier = {
					modifier = party_popularity_dynamic_modifier_coalition
				}
				clear_variable = coalition_pp_gain
				clear_array = coalition_partners_array
			}
			else = {
				remove_from_array = {
					coalition_partners_array = coalition_partner_var_temp
				}
				subtract_from_variable = {
					coalition_pp_gain = party_pop_array^coalition_partner_var_temp
				}
			}
		}
	}
	hidden_effect = {
		set_temp_variable = {
			var = ruling_popularity_var_temp
			value = -0.001
		}
		DSW_FRA_dep_add_ruling_party_popularity = yes
	}
	DSW_FRA_dep_update_party_popularity = yes
}

DSW_FRA_dep_add_to_coalition = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					political_power_ideology_var = coalition_partner_var_temp
				}
				is_in_array = {
					coalition_partners_array = coalition_partner_var_temp
				}
				AND = {
					has_government = var:coalition_partner_var_temp
					has_dynamic_modifier = {
						modifier = party_popularity_dynamic_modifier
					}
				}
			}
		}
		custom_effect_tooltip = add_to_coalition
		hidden_effect = {
			set_temp_variable = {
				party_index = 20
			}
			set_temp_variable = {
				party_popularity_increase = 0
			}
			change_relative_party_popularity = yes
			recalculate_party = yes
		}
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					political_power_ideology_var = coalition_partner_var_temp
				}
				is_in_array = {
					coalition_partners_array = coalition_partner_var_temp
				}
				AND = {
					has_government = var:coalition_partner_var_temp
					has_dynamic_modifier = {
						modifier = party_popularity_dynamic_modifier
					}
				}
			}
		}
		set_variable = {
			is_in_coalition_with_@var:coalition_partner_var_temp = 1
		}
		add_to_array = {
			coalition_partners_array = coalition_partner_var_temp
		}
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = {
							modifier = party_popularity_dynamic_modifier_coalition
						}
					}
				}
				add_dynamic_modifier = {
					modifier = party_popularity_dynamic_modifier_coalition
				}
			}
		}
		add_to_variable = {
			coalition_pp_gain = party_pop_array^coalition_partner_var_temp
		}
	}
	hidden_effect = {
		set_temp_variable = {
			var = ruling_popularity_var_temp
			value = 0.001
		}
		DSW_FRA_dep_add_ruling_party_popularity = yes
	}
	DSW_FRA_dep_update_party_popularity = yes
}

DSW_FRA_dep_update_party_popularity = {
	hidden_effect = {
		if = {
			limit = {
				has_variable = coalition_partners_array^num
			}
			set_variable = {
				coalition_pp_gain = 0
			}
			for_each_loop = {
				array = coalition_partners_array
				add_to_variable = {
					coalition_pp_gain = party_pop_array^v
				}
			}
			multiply_variable = {
				coalition_pp_gain = 0.5
			}
		}
		if = {
			limit = {
				has_variable = ruling_party_wings_array^num
			}
			set_variable = {
				ruling_party_wings_pp_gain = 0
			}
			for_each_loop = {
				array = ruling_party_wings_array
				add_to_variable = {
					ruling_party_wings_pp_gain = party_pop_array^v
				}
			}
			multiply_variable = {
				ruling_party_wings_pp_gain = 0.9
			}
		}
	}
}

DSW_FRA_dep_add_conservative_popularity = {
	if = {
		limit = {
			check_variable = {
				var = conservative_popularity_var_temp
				value = 0
				compare = greater_than
			}
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						party_pop_array^1 < 0.33
					}
					check_variable = {
						party_pop_array^1 > 0.5
					}
				}
			}
			multiply_temp_variable = {
				var = conservative_popularity_var_temp
				value = 0.66
			}
		}
		else_if = {
			limit = {
				NOT = {
					check_variable = {
						party_pop_array^1 < 0.5
					}
					check_variable = {
						party_pop_array^1 > 0.66
					}
				}
			}
			multiply_temp_variable = {
				var = conservative_popularity_var_temp
				value = 0.5
			}
		}
		else_if = {
			limit = {
				NOT = {
					check_variable = {
						party_pop_array^1 < 0.66
					}
					check_variable = {
						party_pop_array^1 > 0.75
					}
				}
			}
			multiply_temp_variable = {
				var = conservative_popularity_var_temp
				value = 0.34
			}
		}
		else_if = {
			limit = {
				NOT = {
					check_variable = {
						party_pop_array^1 < 0.75
					}
				}
			}
			multiply_temp_variable = {
				var = conservative_popularity_var_temp
				value = 0.25
			}
		}
	}
	custom_effect_tooltip = add_conservative_popularity_tooltip
	hidden_effect = {
		set_temp_variable = {
			party_index = 1
		}
		set_temp_variable = {
			party_popularity_increase = conservative_popularity_var_temp

}
