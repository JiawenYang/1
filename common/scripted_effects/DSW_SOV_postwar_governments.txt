DSW_SOV_install_authoritarian_government = {
	if = {
		limit = {
			NOT = {
				has_global_flag = SOV_cprf_won
			}
		}
		create_country_leader = {
			name = DSW_SOV_reconstruction_committee
			picture = "gfx/leaders/SOV/SOV_russian_occupational_administration.png"
			ideology = Neutral_Autocracy
			traits = {
				DSW_SOV_authoritarian_democracy
			}
		}
		DSW_SOV_set_authoritarian_party = yes
		set_party_name = {
			ideology = neutrality
			name = DSW_SOV_reconstruction_front
			long_name = DSW_SOV_reconstruction_front_long
		}
	}
}
