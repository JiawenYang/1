country_event = {
	id = zzz_party.1
	title = zzz_party.1.t
	desc = zzz_party.1.d
	is_triggered_only = no
	trigger = {
		has_global_flag = gaw_china_victory_settled
		original_tag = IND
		NOT = { has_country_flag = zzz_party_ind_done }
	}
	option = {
		name = zzz_party.1.a
		set_country_flag = zzz_party_ind_done
		ZZZ_gaw_gov_IND = yes
		ZZZ_party_leader_IND_tommy_suharto = yes
		add_stability = 0.05
	}
}
