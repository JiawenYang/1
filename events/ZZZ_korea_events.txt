# ============================================================================
#  ZZZ_korea_events —— 朝鲜半岛统一链（本工程自有事件，不再基于基座文件）
#
#  设计口径：
#    1. 事件 ID 全部落在本工程自建段 zzz_kor.*，与基座 korea.* 彻底断开；
#    2. 旗标沿用既有名字（国策/决议/调试工具仍在引用）：
#         korean_confederation_formed   邦联成立（军事同盟，不合并国家）
#         korea_peninsula_reunited      半岛完成合并
#         korean_postwar_settled        战后安排已结算（两条路线互斥）
#         KOR_peninsula_talks_resolved  半岛谈判已落定（三八线决议不再显示）
#         KOR_reunification_talks_failed 谈崩分支已发生
#         zzz_kpr_proclaimed            KPR 已成立
#         zzz_kpr_election_done         NKO 路线全半岛选举已执行
#         zzz_kpr_election_done_kor     KOR 路线全半岛选举已执行
#         zzz_korea_north_annexed       北吞南（NKO 吞并 KOR，军事压倒后触发）
#    3. 事件只做本链自己的事；不调用、不复制任何 korea.* 事件。
#
#  链顺序：
#    zzz_kor.1  谈判开启（KOR 或 NKO 收到；a=继续谈，b=谈崩）
#      ├─ a → zzz_kor.2 谈判成功：成立邦联（korean_confederation_formed）
#      └─ b → zzz_kor.3 谈崩（KOR_reunification_talks_failed）
#    统一落地（两条互斥方向，由战后决议派发）：
#      zzz_kor.4  朝鲜吞并韩国（NKO 统一）
#      zzz_kor.5  韩国吞并朝鲜（KOR 统一，加入太约）
#    KPR 与选举：
#      zzz_kor.6  KPR 成立 → zzz_kor.7 新闻 → 140 天后 zzz_kor.8 全半岛选举
#      zzz_kor.9  选举结果新闻
#      zzz_kor.10 KOR 路线全半岛选举
#    zzz_kor.11 志愿军遗骸归国新闻（原 korea.31 的替代，供年度效果引用）
# ============================================================================

add_namespace = zzz_kor

# .1 谈判开启
#    KOR 收到（中国斡旋决议）或 NKO 收到（韩国重启和谈决议）均可。
#    a 选项要求双方同属民主阵营或同属共产阵营，避免跨阵营强行统一。
# ----------------------------------------------------------------------------
country_event = {
    id = zzz_kor.1
    title = zzz_kor.1.t
    desc = zzz_kor.1.d
    picture = GFX_politics_negotiations
    is_triggered_only = yes

    trigger = {
        NOT = { has_country_flag = korea_peninsula_reunited }
        NOT = { has_global_flag = korean_postwar_settled }
        NOT = { has_global_flag = zzz_korea_north_annexed }
        OR = {
            AND = {
                tag = KOR
                country_exists = NKO
            }
            AND = {
                tag = NKO
                country_exists = KOR
            }
        }
    }

    option = {
        name = zzz_kor.1.a
        trigger = {
            country_exists = KOR
            country_exists = NKO
            OR = {
                AND = {
                    KOR = { has_government = democratic }
                    NKO = { has_government = democratic }
                }
                AND = {
                    KOR = { has_government = communism }
                    NKO = { has_government = communism }
                }
            }
        }
        log = "[GetDateText]: [This.GetName]: zzz_kor.1.a executed"
        add_opinion_modifier = {
            target = KOR
            modifier = imminent_reunification
        }
        add_war_support = -0.05
        KOR = {
            country_event = {
                id = zzz_kor.2
                days = 1
            }
        }
        ai_chance = {
            base = 1
            modifier = {
                add = 1
                KOR = { check_variable = { reunification_chance > 6 } }
            }
            modifier = {
                add = 1
                KOR = { check_variable = { reunification_chance > 8 } }
            }
            modifier = {
                add = 1
                KOR = { check_variable = { reunification_chance > 10 } }
            }
            modifier = {
                factor = 0
                OR = {
                    has_war_with = KOR
                    KOR = { check_variable = { reunification_chance < 6 } }
                }
            }
        }
    }

    option = {
        name = zzz_kor.1.b
        log = "[GetDateText]: [This.GetName]: zzz_kor.1.b executed"
        KOR = {
            set_country_flag = KOR_reunification_talks_failed
            set_country_flag = KOR_peninsula_talks_resolved
            country_event = {
                id = zzz_kor.3
                days = 1
            }
        }
        ai_chance = {
            base = 5
        }
    }
}
