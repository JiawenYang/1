add_namespace = historical_elections

country_event = {
	id = historical_elections.1
	title = historical_elections.1.t
	desc = historical_elections.1.d
	picture = GFX_election_day
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = ITA
		has_country_flag = generic_election_killswitch
		NOT = { has_country_flag = rework_ITA_2018_election }
	}

	immediate = {
		set_country_flag = rework_ITA_2018_election
		set_variable = { anarchist_communism_leader = 2 }
		set_temp_variable = { rul_party_temp = 5 }
		set_temp_variable = { col_one = 6 }
		set_temp_variable = { disable_elections = 0 }
		change_ruling_party_effect = yes
		set_politics = {
			ruling_party = democratic
			last_election = "2018.3.4"
			election_frequency = 60
			elections_allowed = yes
		}
		clr_country_flag = generic_election_killswitch
		set_country_flag = ZZZ_election_enabled
		set_variable = { ZZZ_term = 60 }
		set_variable = { ZZZ_sched_mode = 0 }
		set_variable = { ZZZ_two_round = 0 }
		set_variable = { ZZZ_coalition_ok = 1 }
		set_variable = { ZZZ_leader_mode = 1 }
		set_variable = { ZZZ_grc_bonus = 0 }
		set_variable = { ZZZ_campaign_boost = 0 }
		set_variable = { ZZZ_campaign_boost_opp = 0 }
	}

	option = {
		name = historical_elections.1.a
	}
}
