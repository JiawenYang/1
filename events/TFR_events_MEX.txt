# 毒枭战争爆发
country_event = {
	id = mexico.11
	title = mexico.11.t
	desc = mexico.11.d
	picture = GFX_Generic_Mexican_Police

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = MEX
		NOT = { has_global_flag = MEX_mexican_cartel_war_global }
	}

	immediate = {
		hidden_effect = {
			set_global_flag = MEX_mexican_cartel_war_global
			set_global_flag = flag_mexico_cartel_war

			# 锡那罗亚集团控制西北
			SLA = {
				transfer_state = 828
				load_oob = "cartel_828"
				transfer_state = 829
				load_oob = "cartel_829"
				transfer_state = 830
				load_oob = "cartel_830"
				transfer_state = 831
				load_oob = "cartel_831"
			}

			# 蒂拉卡连特集团控制南部与西部
			TRC = {
				transfer_state = 836
				load_oob = "cartel_836"
				transfer_state = 837
				load_oob = "cartel_837"
				transfer_state = 834
				load_oob = "cartel_834"
			}

			# 塔毛利帕斯集团控制东北与墨西哥湾
			TAM = {
				transfer_state = 832
				load_oob = "cartel_832"
				transfer_state = 833
				load_oob = "cartel_833"
				transfer_state = 838
				load_oob = "cartel_838"
				transfer_state = 839
				load_oob = "cartel_839"
				transfer_state = 840
				load_oob = "cartel_840"
			}

			# 联邦政府权威崩溃，军队大规模倒戈（千禧黎明原版兵力分割机制）
			MEX = {
				add_ideas = MEX_collapsed_authority_tfx
				transfer_units_fraction = {
					target = SLA
					size = 0.25
					stockpile_ratio = 0.25
					army_ratio = 0.25
					navy_ratio = 0.25
					air_ratio = 0.25
				}
				transfer_units_fraction = {
					target = TRC
					size = 0.33
					stockpile_ratio = 0.33
					army_ratio = 0.33
					navy_ratio = 0.33
					air_ratio = 0.33
				}
				transfer_units_fraction = {
					target = TAM
					size = 0.50
					stockpile_ratio = 0.50
					army_ratio = 0.50
					navy_ratio = 0.50
					air_ratio = 0.50
				}
			}

			# 复用千禧黎明原版宣战逻辑：三大集团与联邦政府全面开战
			country_event = {
				id = cartel_event.13
				days = 1
			}
		}
	}

	option = {
		name = mexico.11.a
		add_war_support = 0.10
		add_stability = -0.05
	}
}

# 毒枭战争结束（由 on_actions_ZZZ_peace.txt 结算触发）
country_event = {
	id = mexico.12
	title = mexico.12.t
	desc = mexico.12.d
	picture = GFX_Generic_Mexican_Police

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			tag = MEX
			tag = SLA
			tag = TRC
			tag = TAM
		}
	}

	immediate = {
		hidden_effect = {
			clr_global_flag = MEX_mexican_cartel_war_global
			set_global_flag = MEX_mexican_cartel_war_over_global
			if = {
				limit = {
					tag = MEX
				}
				set_capital = {
					state = 835
					remember_old_capital = no
				}
				remove_ideas = MEX_collapsed_authority_tfx
				add_ideas = MEX_mexican_resconstruction_idea
				country_event = {
					id = mexico.13
					days = 15
				}
			}
		}
	}

	option = {
		name = mexico.12.a
		add_stability = 0.05
		add_political_power = 50
	}
}
