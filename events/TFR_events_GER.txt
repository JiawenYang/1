# Oil Shortage Subsides - 石油短缺消退（循环恢复：每100天自动降级一次石油危机idea）
country_event = {
	id = germany.116
	title = germany.116.t
	desc = germany.116.d
	picture = GFX_GER_fuel_price
	is_triggered_only = yes
	trigger = {
		OR = {
			has_idea = GER_oil_crash1
			has_idea = GER_oil_crash2
			has_idea = GER_oil_crash3
			has_idea = GER_oil_crash4
			has_idea = GER_oil_crash5
		}
	}
	immediate = {
		hidden_effect = {
			log = "[GetDateText]: [Root.GetName]: event germany.116 immediate"
			PRC_decrease_oil = yes
			set_country_flag = GER_finalize_oil_crash_recovery_flag
			# 只要还有石油危机idea，就100天后再触发一次（循环直到彻底恢复）
			if = {
				limit = {
					OR = {
						has_idea = GER_oil_crash1
						has_idea = GER_oil_crash2
						has_idea = GER_oil_crash3
						has_idea = GER_oil_crash4
						has_idea = GER_oil_crash5
					}
				}
				country_event = { id = germany.116 days = 100 }
			}
		}
	}
	option = {
		name = germany.116.a
		unlock_decision_tooltip = GER_finalize_recovery_from_the_oil_crisis
		effect_tooltip = {
			PRC_decrease_oil = yes
		}
	}
}
