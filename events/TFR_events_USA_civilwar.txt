	is_triggered_only = yes
	major = yes
	fire_only_once = yes

	trigger = {
		tag = USA
	}

	immediate = {
		hidden_effect = {
			if = {
				limit = {
					NOT = { is_in_faction_with = JAP }
				}

				}
				set_country_flag = pdto_member
			}
			if = {

					NOT = { has_war_with = CHI }

					target = CHI
					type = annex_everything
				}
			}
			add_named_threat = {
				threat = 10
				name = USA_pacific_mobilization_threat
			}
		}
	}

	option = {
		name = usacw.14.a
		effect_tooltip = {
			if = {
				limit = {
					NOT = { is_in_faction_with = JAP }
				}
				JAP = {
					add_to_faction = USA
				}
			}
			declare_war_on = {
				target = CHI
				type = annex_everything
			}
		}
	}
