# ---- 英国：2024.7.4 大选，工党上台（原生选举 2022.5.1 之后，下一次 2027）----
country_event = {
	id = zzz_party.14
	title = zzz_party.14.t
	desc = zzz_party.14.d
	is_triggered_only = no
	fire_only_once = yes
	trigger = {
		date > 2024.7.5
		original_tag = ENG
	}
	option = {
		name = zzz_party.14.a
		ZZZ_party_to_socialism = yes
		set_leader = yes
	}
}
