# =====================================================================
# 泰国（SIA）—— 君主立宪制选举事件
#
# 口径：
#   1) 选举只改三样：24 槽支持率（party_pop_array^0..23）、执政党标量
#      （set_variable = { ruling_party = N }）、联盟席位数组
#      （gov_coalition_array，只放联盟党，不含执政党）。
#   2) 结尾统一走 startup_politics，不写 recalculate_party。
#   3) 国王由 ZZZ_monarch_SIA_* + ZZZ_monarch_guard 维持；
#      本文件任何事件都不调用 set_leader / kill_country_leader /
#      retire_character / promote_character，也不写 country_leader 角色。
#      2017 历史里的巴育·占奥差（Nat_Autocracy）属于政府首脑岗位，
#      由历史文件建立，本文件不重建、不替换。
#
# 政党槽位：
#   ^1  conservatism        民主党（Prachathipat）
#   ^3  socialism           前进党 / 未来前进党（Move Forward / Future Forward）
#   ^14 Neutral_conservatism 泰自豪党（Bhumjaithai）
#   ^18 neutral_Social      为泰党（Pheu Thai）
#   ^22 Nat_Autocracy       人民国家力量党（Palang Pracharath，军系）
#   ^23 Monarchist          王室派（非选举政党，选举事件中不占票）
#
# 归一规则：只取上述五个参选政党，按席位占比归一为 1.000，
#   其余小党不单列，避免凭空造出不存在的党团。
# =====================================================================

# ---------------------------------------------------------------------
# 2019.03.24 泰国众议院选举（军政府治下举行）
# 现实：为泰党 136 席、人民国家力量党 116 席、未来前进党 81 席、
#       民主党 53 席、泰自豪党 51 席（合计 437 席）。
#       归一：^22 0.265 / ^18 0.312 / ^3 0.185 / ^1 0.121 / ^14 0.117。
# 结果：巴育·占奥差（PPRP）经联盟继续出任总理 → 执政 ^22 nationalist。
# 联盟：民主党 ^1、泰自豪党 ^14（不含执政党 ^22）。
# ---------------------------------------------------------------------
country_event = {
	id = politicsthailand.71
	title = politicsthailand.71.t
	desc = politicsthailand.71.d
	picture = GFX_election_rally
	fire_only_once = yes

	trigger = {
		original_tag = SIA
		date > 2019.03.24
		has_country_flag = ZZZ_monarchy_active
	}

	option = {
		name = politicsthailand.71.a

		set_popularities = {
			democratic = 30.6
			communism = 0.0
			fascism = 0.0
			neutrality = 42.9
			nationalist = 26.5
		}
		set_politics = {
			ruling_party = nationalist
			last_election = "2019.03.24"
			election_frequency = 48
			elections_allowed = yes
		}

		start_politics_input = yes

		### Party Popularities
		set_variable = { party_pop_array^0 = 0 } #Western_Autocracy
		set_variable = { party_pop_array^1 = 0.121 } #conservatism（民主党）
		set_variable = { party_pop_array^2 = 0 } #liberalism
		set_variable = { party_pop_array^3 = 0.185 } #socialism（未来前进党）
		set_variable = { party_pop_array^4 = 0 } #communist_state
		set_variable = { party_pop_array^5 = 0 } #anarchist_communism
		set_variable = { party_pop_array^6 = 0 } #Conservative
		set_variable = { party_pop_array^7 = 0 } #Autocracy
		set_variable = { party_pop_array^8 = 0 } #Mod_Vilayat_e_Faqih
		set_variable = { party_pop_array^9 = 0 } #Vilayat_e_Faqih
		set_variable = { party_pop_array^10 = 0 } #Kingdom
		set_variable = { party_pop_array^11 = 0 } #Caliphate
		set_variable = { party_pop_array^12 = 0 } #Neutral_Muslim_Brotherhood
		set_variable = { party_pop_array^13 = 0 } #Neutral_Autocracy
		set_variable = { party_pop_array^14 = 0.117 } #Neutral_conservatism（泰自豪党）
		set_variable = { party_pop_array^15 = 0 } #oligarchism
		set_variable = { party_pop_array^16 = 0 } #Neutral_Libertarian
		set_variable = { party_pop_array^17 = 0 } #Neutral_green
		set_variable = { party_pop_array^18 = 0.312 } #neutral_Social（为泰党）
		set_variable = { party_pop_array^19 = 0 } #Neutral_Communism
		set_variable = { party_pop_array^20 = 0 } #Nat_Populism
		set_variable = { party_pop_array^21 = 0 } #Nat_Fascism
		set_variable = { party_pop_array^22 = 0.265 } #Nat_Autocracy（人民国家力量党）
		set_variable = { party_pop_array^23 = 0 } #Monarchist

		### Ruling Party
		set_variable = { ruling_party = 22 }
		add_to_array = { gov_coalition_array = 1 }
		add_to_array = { gov_coalition_array = 14 }

		startup_politics = yes
		log = "[GetDateText]: [This.GetName]: politicsthailand.71.a executed (2019 general election, ruling=22)"
	}
}

# ---------------------------------------------------------------------
# 周期选举（2023.05.14 之后的每一届）
# 由 common/on_actions/00_on_actions.txt 的 on_new_term_election 路由：
#   original_tag = SIA + ZZZ_monarchy_active -> politicsthailand.73
# 不做历史数值注入，按当前 24 槽支持率计票（hold_elections）后再走同一套
# 三件套写入；国王不参与，也不碰任何元首角色。
# ---------------------------------------------------------------------
country_event = {
	id = politicsthailand.73
	hidden = yes
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			log = "[GetDateText]: [This.GetName]: politicsthailand.73 executed (cycle election, chain=SIA_special)"
			hold_elections = yes

			set_temp_variable = { rul_party_temp = max_index_1 }
			set_variable = { ruling_party = rul_party_temp }

			clear_array = gov_coalition_array
			if = {
				limit = {
					NOT = { check_variable = { party_pop_array^ruling_party > majority_threshold } }
					has_variable = max_index_2
				}
				set_temp_variable = { col_one = max_index_2 }
				add_to_array = { gov_coalition_array = col_one }
			}

			update_set_politics = yes
			startup_politics = yes
		}
	}
}
