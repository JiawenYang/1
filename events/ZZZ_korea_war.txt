# 2022.5.9 谈判窗口关闭：只做分叉，不宣战、不升温
country_event = {
	id = ZZZ_KW_korea_war.1
	title = ZZZ_KW_korea_war.1.t
	desc = ZZZ_KW_korea_war.1.d
	picture = GFX_politics_negotiations
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		tag = KOR
		date > 2022.5.8
	}

	option = {
		name = ZZZ_KW_korea_war.1.a
		log = "[GetDateText]: [This.GetName]: ZZZ_KW_korea_war.1.a executed"
		if = {
			limit = {
				OR = {
					has_global_flag = korean_confederation_formed
					has_country_flag = korea_peninsula_reunited
				}
			}
			set_country_flag = ZZZ_kw_peace_settled
			country_event = { id = ZZZ_KW_korea_war.2 days = 1 }
		}
		else = {
			set_country_flag = ZZZ_kw_tension_branch
			country_event = { id = ZZZ_KW_korea_war.3 days = 1 }
		}
	}
}

# 2024.12.3 戒严尝试：成败只看「台湾是否已统一」
#   台湾已统一（特别行政区落地）→ 戒严成功 → .5
#   台湾未统一                    → 戒严失败 → .9
country_event = {
	id = ZZZ_KW_korea_war.4
	title = ZZZ_KW_korea_war.4.t
	desc = ZZZ_KW_korea_war.4.d
	picture = GFX_politics_talks
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		tag = KOR
		date > 2024.11.1
		has_country_flag = ZZZ_kw_conservative_government
		NOT = { has_country_flag = ZZZ_kw_martial_law_started }
		NOT = { has_country_flag = ZZZ_kw_martial_law_failed }
	}

	option = {
		name = ZZZ_KW_korea_war.4.a
		log = "[GetDateText]: [This.GetName]: ZZZ_KW_korea_war.4.a executed"
		if = {
			limit = { ZZZ_taiwan_is_sar = yes }
			set_country_flag = ZZZ_kw_martial_law_started
			add_ideas = ZZZ_KW_martial_law_2024
			country_event = { id = ZZZ_KW_korea_war.5 days = 2 }
		}
		else = {
			set_country_flag = ZZZ_kw_martial_law_failed
			country_event = { id = ZZZ_KW_korea_war.9 days = 1 }
		}
	}
}
